cmake_minimum_required(VERSION 3.10)
project(RandomVector LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(random_vector main.cpp random_vector.cpp)
add_executable(random_vector_test test_random_vector.cpp random_vector.cpp)
enable_testing()
add_test(NAME random_vector_test COMMAND random_vector_test)
if(MSVC)
  target_compile_options(random_vector PRIVATE /W4)
  target_compile_options(random_vector_test PRIVATE /W4)
else()
  target_compile_options(random_vector PRIVATE -Wall -Wextra -pedantic)
  target_compile_options(random_vector_test PRIVATE -Wall -Wextra -pedantic)
endif()
